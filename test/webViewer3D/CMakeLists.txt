cmake_minimum_required(VERSION 3.18)
project(WebSelectableOctGridTests LANGUAGES CXX)
find_package(Boost 1.70 CONFIG REQUIRED)
find_package(Eigen3 5 REQUIRED CONFIG)
find_package(glog REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_executable(viewer_fixture fixture.cpp ../../src/Net/HttpServer.cpp ../../src/IO/WebSocketStream.cpp)
target_include_directories(viewer_fixture PRIVATE ${Boost_INCLUDE_DIRS})
target_compile_definitions(viewer_fixture PRIVATE BOOST_ERROR_CODE_HEADER_ONLY USE_GLOG)
target_link_libraries(viewer_fixture PRIVATE Threads::Threads Eigen3::Eigen glog::glog)
enable_testing()
add_executable(grid_protocol grid_protocol.cpp)
target_compile_definitions(grid_protocol PRIVATE USE_GLOG)
target_link_libraries(grid_protocol PRIVATE Eigen3::Eigen glog::glog)
target_compile_options(grid_protocol PRIVATE -UNDEBUG)
add_test(NAME grid_protocol COMMAND grid_protocol)
add_test(NAME webviewer_transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/integration.py
  $<TARGET_FILE:viewer_fixture> ${CMAKE_CURRENT_SOURCE_DIR}/../../html/viewer/_SelectableOctGrid)
set_tests_properties(webviewer_transport PROPERTIES TIMEOUT 30)
