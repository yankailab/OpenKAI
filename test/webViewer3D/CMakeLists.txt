cmake_minimum_required(VERSION 3.18)
project(WebViewer3DTests LANGUAGES CXX)
find_package(Boost 1.70 REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_executable(viewer_fixture fixture.cpp ../../src/Net/HttpServer.cpp ../../src/IO/WebSocketStream.cpp)
target_include_directories(viewer_fixture PRIVATE ${Boost_INCLUDE_DIRS})
target_compile_definitions(viewer_fixture PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(viewer_fixture PRIVATE Threads::Threads)
enable_testing()
add_test(NAME webviewer_transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/integration.py
  $<TARGET_FILE:viewer_fixture> ${CMAKE_CURRENT_SOURCE_DIR}/../../html/webViewer3D)
set_tests_properties(webviewer_transport PROPERTIES TIMEOUT 30)
