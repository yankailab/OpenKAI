cmake_minimum_required(VERSION 3.18)
project(SLAMTests LANGUAGES CXX)

find_package(Eigen3 5 REQUIRED CONFIG)
find_package(glim REQUIRED CONFIG)
find_package(glog REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)
set(BUILD_AS_LIB ON CACHE BOOL "" FORCE)
set(WITH_SLAM ON CACHE BOOL "" FORCE)
set(WITH_UNIVERSE ON CACHE BOOL "" FORCE)
set(USE_GLIM ON CACHE BOOL "" FORCE)
add_subdirectory(../.. openkai)
add_executable(slam_integration integration.cpp)
target_compile_features(slam_integration PRIVATE cxx_std_17)
target_compile_options(slam_integration PRIVATE -UNDEBUG)
target_compile_definitions(slam_integration PRIVATE USE_GLOG WITH_SLAM USE_GLIM)
target_link_libraries(slam_integration PRIVATE OpenKAI Eigen3::Eigen glim::glim glog::glog)
enable_testing()
add_test(NAME slam_integration COMMAND slam_integration
  ${CMAKE_CURRENT_SOURCE_DIR}/../../jsonCfg/glim_orbbec ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(slam_integration PROPERTIES TIMEOUT 90)

add_executable(slam_imu_free imu_free.cpp)
target_compile_features(slam_imu_free PRIVATE cxx_std_17)
target_compile_options(slam_imu_free PRIVATE -UNDEBUG)
target_compile_definitions(slam_imu_free PRIVATE USE_GLOG WITH_SLAM USE_GLIM)
target_link_libraries(slam_imu_free PRIVATE OpenKAI Eigen3::Eigen glim::glim glog::glog)
add_test(NAME slam_imu_free COMMAND slam_imu_free
  ${CMAKE_CURRENT_SOURCE_DIR}/../../jsonCfg/glim_scepter ${CMAKE_CURRENT_BINARY_DIR}/imu_free_fixtures)
set_tests_properties(slam_imu_free PROPERTIES TIMEOUT 90)

add_executable(pcfile_test pcfile_test.cpp)
target_compile_features(pcfile_test PRIVATE cxx_std_17)
target_compile_options(pcfile_test PRIVATE -UNDEBUG)
target_compile_definitions(pcfile_test PRIVATE USE_GLOG)
target_link_libraries(pcfile_test PRIVATE OpenKAI Eigen3::Eigen glog::glog)
add_test(NAME pcfile_test COMMAND pcfile_test)

add_executable(webglim_stream_fixture webglim_stream_fixture.cpp)
target_compile_features(webglim_stream_fixture PRIVATE cxx_std_17)
target_compile_definitions(webglim_stream_fixture PRIVATE BOOST_ERROR_CODE_HEADER_ONLY USE_GLOG WITH_SLAM USE_GLIM)
target_link_libraries(webglim_stream_fixture PRIVATE OpenKAI Eigen3::Eigen Threads::Threads glog::glog)
add_test(NAME webglim_stream COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/webglim_stream.py
  $<TARGET_FILE:webglim_stream_fixture> ${CMAKE_CURRENT_SOURCE_DIR}/../../html/viewer/_GLIM)
set_tests_properties(webglim_stream PROPERTIES TIMEOUT 30)
