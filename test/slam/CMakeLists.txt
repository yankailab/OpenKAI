cmake_minimum_required(VERSION 3.18)
project(SLAMTests LANGUAGES CXX)

find_package(Eigen3 5 REQUIRED CONFIG)
find_package(glim REQUIRED CONFIG)
find_package(glog REQUIRED)
set(BUILD_AS_LIB ON CACHE BOOL "" FORCE)
set(WITH_SLAM ON CACHE BOOL "" FORCE)
set(USE_GLIM ON CACHE BOOL "" FORCE)
add_subdirectory(../.. openkai)
add_executable(slam_integration integration.cpp)
target_compile_features(slam_integration PRIVATE cxx_std_17)
target_compile_options(slam_integration PRIVATE -UNDEBUG)
target_compile_definitions(slam_integration PRIVATE USE_GLOG WITH_SLAM USE_GLIM)
target_link_libraries(slam_integration PRIVATE OpenKAI Eigen3::Eigen glim::glim glog::glog)
enable_testing()
add_test(NAME slam_integration COMMAND slam_integration
  ${CMAKE_CURRENT_SOURCE_DIR}/../../jsonCfg/glim_orbbec ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(slam_integration PROPERTIES TIMEOUT 90)
